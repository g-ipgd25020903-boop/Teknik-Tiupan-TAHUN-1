    <script>

        // Sistem Bunyi Sintesis (Web Audio API) untuk maklum balas mesra kanak-kanak

        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();


        function playSound(type) {

            if (audioCtx.state === 'suspended') {

                audioCtx.resume();

            }

            const osc = audioCtx.createOscillator();

            const gainNode = audioCtx.createGain();

            osc.connect(gainNode);

            gainNode.connect(audioCtx.destination);


            if (type === 'correct') {

                // Bunyi gembira jika betul

                osc.type = 'triangle';

                osc.frequency.setValueAtTime(300, audioCtx.currentTime);

                osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.3);

                gainNode.gain.setValueAtTime(0.15, audioCtx.currentTime);

                gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);

                osc.start();

                osc.stop(audioCtx.currentTime + 0.3);

            } else if (type === 'wrong') {

                // Bunyi kecewa lembut jika salah

                osc.type = 'sawtooth';

                osc.frequency.setValueAtTime(220, audioCtx.currentTime);

                osc.frequency.linearRampToValueAtTime(110, audioCtx.currentTime + 0.4);

                gainNode.gain.setValueAtTime(0.1, audioCtx.currentTime);

                gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);

                osc.start();

                osc.stop(audioCtx.currentTime + 0.4);

            } else if (type === 'click') {

                // Bunyi klik menu

                osc.type = 'sine';

                osc.frequency.setValueAtTime(440, audioCtx.currentTime);

                gainNode.gain.setValueAtTime(0.1, audioCtx.currentTime);

                gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);

                osc.start();

                osc.stop(audioCtx.currentTime + 0.1);

            } else if (type === 'blow') {

                // Bunyi desiran tiupan

                const bufferSize = audioCtx.sampleRate * 0.4;

                const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);

                const data = buffer.getChannelData(0);

                for (let i = 0; i < bufferSize; i++) {

                    data[i] = Math.random() * 2 - 1;

                }

                const noise = audioCtx.createBufferSource();

                noise.buffer = buffer;

                const filter = audioCtx.createBiquadFilter();

                filter.type = 'bandpass';

                filter.frequency.value = 800;

                noise.connect(filter);

                filter.connect(gainNode);

                gainNode.gain.setValueAtTime(0.05, audioCtx.currentTime);

                gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);

                noise.start();

            } else if (type === 'win') {

                // Melodi kemenangan

                const notes = [261.63, 329.63, 392.00, 523.25]; // C, E, G, C

                notes.forEach((freq, index) => {

                    const oscNote = audioCtx.createOscillator();

                    const gainNote = audioCtx.createGain();

                    oscNote.connect(gainNote);

                    gainNote.connect(audioCtx.destination);

                    oscNote.type = 'sine';

                    oscNote.frequency.value = freq;

                    gainNote.gain.setValueAtTime(0.1, audioCtx.currentTime + index * 0.15);

                    gainNote.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + index * 0.15 + 0.2);

                    oscNote.start(audioCtx.currentTime + index * 0.15);

                    oscNote.stop(audioCtx.currentTime + index * 0.15 + 0.2);

                });

            }

        }

        function redrawAll() {

            // Bersihkan kanvas putih

            ctx.clearRect(0, 0, canvas.width, canvas.height);

            ctx.fillStyle = '#fafafa';

            ctx.fillRect(0, 0, canvas.width, canvas.height);


            // Lukis latar belakang hiasan grid seni bertitik

            ctx.fillStyle = '#f1f5f9';

            for (let i = 0; i < canvas.width; i += 20) {

                for (let j = 0; j < canvas.height; j += 20) {

                    ctx.fillRect(i, j, 2, 2);

                }

            }


            // Lukis dahan/jalur tiupan cat air yang telah ditiup dahulu

            branches.forEach(branch => {

                ctx.strokeStyle = branch.color;

                ctx.lineWidth = branch.width;

                ctx.lineCap = 'round';

                ctx.lineJoin = 'round';

                ctx.beginPath();

                ctx.moveTo(branch.x1, branch.y1);

                ctx.lineTo(branch.x2, branch.y2);

                ctx.stroke();

            });


            // Lukis titisan-titisan cat bulat basah yang sedia ditiup

            droplets.forEach(drop => {

                ctx.fillStyle = drop.color;

                ctx.beginPath();

                ctx.arc(drop.x, drop.y, drop.radius, 0, Math.PI * 2);

                ctx.fill();

            });


            // Sentiasa lukis template di atas dahan & titisan agar nampak menyatu

            drawTemplateOutline();

        }


        // Mendapatkan posisi input (mouse / touch)

        function getCoords(e) {

            const rect = canvas.getBoundingClientRect();

            let clientX, clientY;

            if (e.touches && e.touches.length > 0) {

                clientX = e.touches[0].clientX;

                clientY = e.touches[0].clientY;

            } else {

                clientX = e.clientX;

                clientY = e.clientY;

            }

            return {

                x: (clientX - rect.left) * (canvas.width / rect.width),

                y: (clientY - rect.top) * (canvas.height / rect.height)

            };

        }


        // Mula sentuh/klik: menitiskan cat atau bersedia meniup

        function handleStart(e) {

            e.preventDefault();

            const pos = getCoords(e);

            

            // Cari jika kursor dekat dengan titisan sedia ada untuk 'meniup'

            let nearDrop = false;

            droplets.forEach(drop => {

                const dist = Math.hypot(drop.x - pos.x, drop.y - pos.y);

                if (dist < 30) {

                    nearDrop = true;

                }

            });


            if (!nearDrop) {

                // Titiskan cat baharu di sini

                droplets.push({

                    x: pos.x,

                    y: pos.y,

                    radius: 8 + Math.random() * 5,

                    color: selectedColor

                });

                playSound('click');

                redrawAll();

            }


            isDrawing = true;

            // Rekod posisi mula tiupan

            this.lastX = pos.x;

            this.lastY = pos.y;

        }


        function handleMove(e) {

            if (!isDrawing) return;

            e.preventDefault();

            const pos = getCoords(e);


            const dx = pos.x - this.lastX;

            const dy = pos.y - this.lastY;

            const dist = Math.hypot(dx, dy);


            // Jika kelajuan drag mencukupi (umpama hembusan angin fizikal)

            if (dist > 5) {

                // Kesan tiupan ke atas mana-mana titisan yang berhampiran dengan permulaan leretan

                droplets.forEach((drop, index) => {

                    const toDropDist = Math.hypot(drop.x - this.lastX, drop.y - this.lastY);

                    if (toDropDist < 25) {

                        // Tiup titisan ke hadapan!

                        const blowAngle = Math.atan2(dy, dx);

                        const blowLength = 20 + Math.random() * 30; // Kepanjangan dahan tiupan

                        

                        const targetX = drop.x + Math.cos(blowAngle) * blowLength;

                        const targetY = drop.y + Math.sin(blowAngle) * blowLength;


                        // Rekod dahan cat terhasil

                        branches.push({

                            x1: drop.x,

                            y1: drop.y,

                            x2: targetX,

                            y2: targetY,

                            color: drop.color,

                            width: drop.radius

                        });


                        // Tambah beberapa dahan kecil serong (splatter kesan tiupan realistik)

                        const smallBranchCount = 2 + Math.floor(Math.random() * 3);

                        for (let i = 0; i < smallBranchCount; i++) {

                            const sideAngle = blowAngle + (Math.random() - 0.5) * 1.2;

                            const sideLength = blowLength * (0.3 + Math.random() * 0.5);

                            branches.push({

                                x1: drop.x + Math.cos(blowAngle) * (blowLength * 0.4),

                                y1: drop.y + Math.sin(blowAngle) * (blowLength * 0.4),

                                x2: drop.x + Math.cos(sideAngle) * sideLength,

                                y2: drop.y + Math.sin(sideAngle) * sideLength,

                                color: drop.color,

                                width: drop.radius * 0.5

                            });

                        }


                        // Titisan itu berpindah ke hujung dahan baharu tetapi mengecil sedikit

                        drop.x = targetX;

                        drop.y = targetY;

                        drop.radius *= 0.7; // Semakin ditiup semakin kurang cat


                        // Padam jika titisan terlalu kecil

                        if (drop.radius < 2) {

                            droplets.splice(index, 1);

                        }


                        playSound('blow');

                        triggerVisualWind(pos.x, pos.y);

                    }

                });


                redrawAll();

                this.lastX = pos.x;

                this.lastY = pos.y;

            }

        }


        function handleEnd() {

            isDrawing = false;

        }

        async function toggleMic(checkbox) {

            isMicEnabled = checkbox.checked;

            const meter = document.getElementById('mic-meter');


            if (isMicEnabled) {

                try {

                    micStream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false });

                    audioContextMic = new (window.AudioContext || window.webkitAudioContext)();

                    const source = audioContextMic.createMediaStreamSource(micStream);

                    analyser = audioContextMic.createAnalyser();

                    analyser.fftSize = 256;

                    source.connect(analyser);

                    

                    detectBlowingNoise();

                } catch (err) {

                    console.error("Gagal mendapat kebenaran mikrofon:", err);

                    checkbox.checked = false;

                    isMicEnabled = false;

                    meter.style.width = '0%';

                }

            } else {

                if (micStream) {

                    micStream.getTracks().forEach(track => track.stop());

                }

                if (audioContextMic) {

                    audioContextMic.close();

                }

                meter.style.width = '0%';

            }

        }


        // Mengesan aras bunyi mic untuk menggerakkan cat (simulasi tiupan mulut)

        function detectBlowingNoise() {

            if (!isMicEnabled) return;

            

            const dataArray = new Uint8Array(analyser.frequencyBinCount);

            analyser.getByteFrequencyData(dataArray);


            // Hitung purata kelantangan frekuensi rendah & tinggi (ciri tiupan angin berdesir)

            let totalVolume = 0;

            for (let i = 0; i < dataArray.length; i++) {

                totalVolume += dataArray[i];

            }

            const averageVolume = totalVolume / dataArray.length;


            // Kemas kini meter kelantangan di UI

            const meter = document.getElementById('mic-meter');

            const meterWidth = Math.min(averageVolume * 1.5, 100);

            meter.style.width = meterWidth + '%';


            // Jika bunyi hembusan kuat dikesan (ambang > 35)

            if (averageVolume > 35) {

                // Cari mana-mana titisan rawak untuk ditiup rawak ke atas

                if (droplets.length > 0) {

                    const randomIdx = Math.floor(Math.random() * droplets.length);

                    const drop = droplets[randomIdx];


                    // Tiup ke arah atas-kiri/atas-kanan secara rawak (seperti hembusan semula jadi)

                    const angle = -Math.PI/2 + (Math.random() - 0.5) * 1.5;

                    const blowLength = 25 + Math.random() * 20;


                    const targetX = drop.x + Math.cos(angle) * blowLength;

                    const targetY = drop.y + Math.sin(angle) * blowLength;


                    branches.push({

                        x1: drop.x,

                        y1: drop.y,

                        x2: targetX,

                        y2: targetY,

                        color: drop.color,

                        width: drop.radius

                    });


                    drop.x = targetX;

                    drop.y = targetY;

                    drop.radius *= 0.85;


                    if (drop.radius < 2) {

                        droplets.splice(randomIdx, 1);

                    }

                    playSound('blow');

                    triggerVisualWind(drop.x, drop.y);

                    redrawAll();

                }

            }


            requestAnimationFrame(detectBlowingNoise);

        }

        let activeTouchElement = null;

        function handleTouchStart(e) {

            activeTouchElement = this;

            this.classList.add('scale-105', 'shadow-lg');

        }


        function handleTouchMove(e) {

            if (!activeTouchElement) return;

            e.preventDefault(); // Menahan tatal skrin (prevent scrolling) semasa mengheret

            

            const touch = e.touches[0];

            // Visual kedudukan terapung elemen semasa diheret

            activeTouchElement.style.position = 'fixed';

            activeTouchElement.style.left = (touch.clientX - activeTouchElement.offsetWidth / 2) + 'px';

            activeTouchElement.style.top = (touch.clientY - activeTouchElement.offsetHeight / 2) + 'px';

            activeTouchElement.style.zIndex = '1000';

        }


        function handleTouchEnd(e) {

            if (!activeTouchElement) return;

            activeTouchElement.classList.remove('scale-105', 'shadow-lg');

            

            // Cari zon drop paling hampir di bawah jari pelepas

            const touch = e.changedTouches[0];

            activeTouchElement.style.position = '';

            activeTouchElement.style.left = '';

            activeTouchElement.style.top = '';

            activeTouchElement.style.zIndex = '';


            const elementBelow = document.elementFromPoint(touch.clientX, touch.clientY);

            const zone = elementBelow ? elementBelow.closest('.drop-zone') : null;


            if (zone) {

                placeElementInZone(zone, activeTouchElement);

            } else {

                // Hantar balik ke container asal jika diletak luar zon

                document.getElementById('drag-items-container').appendChild(activeTouchElement);

            }

            activeTouchElement = null;

        }
